def balanced_pair(elements: list[int], harmony_target: int) -> tuple[int, int]:

left, right = 0, len(elements) - 1

while left < right:

current_sum = elements[left] + elements[right]

if current_sum == harmony_target:

return (left, right) # Flow state achieved ✨

elif current_sum < harmony_target:

left += 1

else:

right -= 1

return (-1, -1)
